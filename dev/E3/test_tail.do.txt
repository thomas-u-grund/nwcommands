* ---- (3) native change statistics == brute-force statistic differences ----
* (change toward one of every dyad, from the native MPLE design builder, against stat(dyad=1) - stat(dyad=0)
*  computed by the observed-statistic routine; random 7-node graphs.  degcor and tripercent are excluded: R's change
*  statistic of those two is not the difference of its summary statistic (degcor is standardised by the observed
*  variance; tripercent is a sum of single-step ratios).)
mata:
real scalar ns_t_brute(real scalar tol)
{
	external class ErgmModel scalar __nwergm_last_M
	external class ErgmGraph scalar __nwergm_last_G
	real matrix D
	real scalar n, i, j, row, p, maxdiff, tied
	real rowvector s0, s1, ch
	n = __nwergm_last_G.n
	(void) ErgmNativeSetup(__nwergm_last_M, 2, __nwergm_last_G)
	if (!__nwergm_last_M.native_enabled) return(-1)
	D = ErgmNativeBuildMPLEData(__nwergm_last_M, __nwergm_last_G)
	p = __nwergm_last_M.nparam()
	maxdiff = 0
	row = 0
	for (i=1; i<=n; i++) for (j=1; j<=n; j++) {
		if (i == j) continue
		if (!__nwergm_last_G.directed & j < i) continue
		row++
		tied = __nwergm_last_G.has_edge(i, j)
		if (tied) {
			s1 = __nwergm_last_M.full_statistic(__nwergm_last_G)
			__nwergm_last_G.toggle(i, j)
			s0 = __nwergm_last_M.full_statistic(__nwergm_last_G)
			__nwergm_last_G.toggle(i, j)
		}
		else {
			s0 = __nwergm_last_M.full_statistic(__nwergm_last_G)
			__nwergm_last_G.toggle(i, j)
			s1 = __nwergm_last_M.full_statistic(__nwergm_last_G)
			__nwergm_last_G.toggle(i, j)
		}
		ch = s1 - s0
		maxdiff = max((maxdiff, max(abs(ch - D[row, 1..p]))))
	}
	return(maxdiff)
}
end

capture program drop ns_brute
program ns_brute
	args id dir terms
	* random 7-node graph, node attribute a in 1..3
	clear
	qui set obs 7
	qui gen a = 1 + mod(_n * 5, 3)
	matrix A_br = J(7,7,0)
	forvalues i = 1/7 {
		forvalues j = 1/7 {
			if `i' != `j' & runiform() < 0.35 {
				matrix A_br[`i',`j'] = 1
				if !`dir' matrix A_br[`j',`i'] = 1
			}
		}
	}
	ns_stat br `dir' "`terms'"
	mata: st_numscalar("__ns_md", ns_t_brute(1e-9))
	if scalar(__ns_md) < 0 {
		di as err "FAIL brute `id': model not native-eligible"
		exit 9
	}
	if scalar(__ns_md) > 1e-9 {
		di as err "FAIL brute `id': native change statistic differs from the difference of the observed statistic by " scalar(__ns_md)
		exit 9
	}
end
